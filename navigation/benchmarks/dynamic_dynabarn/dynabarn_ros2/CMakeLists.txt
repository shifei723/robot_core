cmake_minimum_required(VERSION 3.16)
project(dynabarn_ros2)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_common_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-common5 REQUIRED COMPONENTS graphics)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(dynabarn_motion SHARED src/motion.cpp)
target_link_libraries(dynabarn_motion PRIVATE gz-sim8::gz-sim8 gz-common5::graphics gz-plugin2::register)
install(TARGETS dynabarn_motion LIBRARY DESTINATION lib)
install(DIRECTORY launch worlds config comparison_config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/evaluate.py scripts/convert_world.py scripts/control_checks.py scripts/laser_perception.py scripts/compare_evaluate.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/localization_profiles.py scripts/map_geometry.py scripts/extract_trajectory.py scripts/benchmark_protocol.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.txt DESTINATION share/${PROJECT_NAME})
install(PROGRAMS ../mpc_d_cbf/node.py DESTINATION lib/${PROJECT_NAME} RENAME mpc_d_cbf_node.py)
install(FILES ../mpc_d_cbf/core.py DESTINATION lib/${PROJECT_NAME} RENAME mpc_d_cbf_core.py)
install(FILES ../mpc_d_cbf/upstream/local_planner.py DESTINATION lib/${PROJECT_NAME} RENAME mpc_d_cbf_upstream.py)
ament_package()
